# 循环

## for 循环

```shell
# 遍历列表
for i in 1 2 3; do
  echo $i
done

# 遍历脚本参数
for i in $@; do
  echo $i
done

# 遍历文件
for f in /etc/*.conf; do
  echo $f
done

# C 风格（bash 专属）
for ((i=0; i<5; i++)); do
  echo $i
done
```

## while 循环

```shell
i=0
while [ $i -lt 10 ]; do
  echo $i
  ((i++))
done

# 读取文件每一行
while read line; do
  echo "$line"
done < file.txt

# 无限循环
while true; do
  echo "running"
  sleep 1
done
```

## until 循环

条件为假时执行，与 while 相反。

```shell
i=0
until [ $i -ge 5 ]; do
  echo $i
  ((i++))
done
```

## 循环控制

```shell
break       # 跳出循环
continue    # 跳过本次，继续下一次
break 2     # 跳出 2 层嵌套循环
```
